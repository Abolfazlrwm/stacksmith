# Chat API — POST a question, get an answer grounded in your documents.
#
# Flow:
#   1. Receive {question, conversation_id?, top_k?}
#   2. Retrieve top_k chunks for the question (RAG over pgvector)
#   3. Pass context to the LLM adapter (OpenAI / Anthropic / Ollama)
#   4. Persist the user message and the assistant answer (with citations)
#   5. Return {answer, citations, conversation_id, message_id, model}

class ChatView(APIView):
    """POST /api/chat/ — ask docubot a question."""

    def post(self, request: HttpRequest) -> Response:
        # Owner scoping — never mix users' corpora.
        owner = _owner_from_request(request)

        ser = ChatRequestSerializer(data=request.data)
        ser.is_valid(raise_exception=True)
        question = ser.validated_data["question"]
        top_k = ser.validated_data["top_k"]

        # 1. Retrieve the most relevant chunks for this owner's corpus.
        hits = retrieve(question, owner=owner, top_k=top_k)

        # 2. Ask the LLM, grounded in the retrieved passages.
        passages = [h.text for h in hits]
        llm_resp = llm_answer(question, passages, owner=owner)

        # 3. Map [1]/[2]/... back to the actual chunks for citations.
        citations = _resolve_citations(llm_resp.text, hits)

        # 4. Persist the conversation turn.
        with transaction.atomic():
            conv = Conversation.objects.create(
                owner=owner, title=question[:80],
            )
            assistant_msg = Message.objects.create(
                conversation=conv,
                role=Message.Role.ASSISTANT,
                content=llm_resp.text,
                citations=citations,
                retrieved_count=len(hits),
                model=llm_resp.model,
            )

        # 5. Shape the response.
        return Response({
            "answer": llm_resp.text,
            "citations": citations,
            "retrieved_count": len(hits),
            "model": llm_resp.model,
        }, status=status.HTTP_200_OK)

"""Retrieval — find the most relevant Chunks for a query.
Uses pgvector's cosine-distance operator (`<=>`) for ANN search
over the HNSW index created in the documents migration."""

def retrieve(query: str, *, owner: str = "",
             top_k: int = 5, min_score: float = 0.20):
    """Find the top_k chunks most similar to `query`, restricted to `owner`.
    `owner` is critical for multi-user isolation: user A must not
    retrieve from user B's documents."""
    from .embeddings import embed_texts
    if not query.strip():
        return []

    # Embed the query with the same model used for chunks.
    qvec = embed_texts([query])[0]

    # pgvector cosine-distance search, restricted to this owner.
    qs = (
        Chunk.objects
        .select_related("document")
        .filter(document__status=Document.Status.READY)
    )
    if owner:
        qs = qs.filter(document__owner=owner)

    # Ordered by cosine distance ascending (most similar first).
    qs = qs.annotate(distance=CosineDistance("embedding", qvec)) \
          .order_by("distance")[:top_k]

    hits = []
    for chunk in qs:
        sim = 1.0 - float(chunk.distance)  # dist→similarity
        if sim < min_score:
            continue  # drop weak matches
        hits.append(RetrievalHit(
            chunk_id=chunk.id,
            document_title=chunk.document.title,
            text=chunk.text,
            score=sim,
        ))
    return hits

def validate_upload(filename: str, data: bytes) -> str:
    """Reject disallowed uploads before any bytes are written to disk.
    Three layers of defense, all run before the upload view writes:

    1. Extension allow-list — accepted suffixes only.
    2. Magic-byte sniff — for PDF, first bytes must match.
    3. UTF-8 decode check — for text/markdown kinds.
    """
    name = filename.lower()
    if not any(name.endswith(ext) for ext in ALLOWED_EXTENSIONS):
        raise ValueError(
            f"unsupported file type: {filename!r}; "
            f"accepted: {sorted(ALLOWED_EXTENSIONS)}"
        )

    kind = detect_kind(filename, mime=None)

    sig = MAGIC_SIGNATURES.get(kind)
    if sig is not None:
        # Layer 2 — magic byte sniff (PDF = "%PDF-")
        if not data.startswith(sig):
            raise ValueError(
                f"file {filename!r} claims to be a {kind} "
                f"but content does not match magic bytes"
            )
    else:
        # Layer 3 — text/markdown: must decode as UTF-8
        try:
            data.decode("utf-8")
        except UnicodeDecodeError as exc:
            raise ValueError(
                f"file {filename!r} is not valid UTF-8 "
                text; refusing upload"
            ) from exc
    return kind
